// SeeNora v3 simplified MVP data model
// Target: PostgreSQL on Desktop/Site and a compatible subset in Room/SQLite.
// Raw vibration samples are immutable files/blobs; only their metadata and
// integrity hashes are stored in this database.

Enum asset_node_type {
  LINE
  DEPARTMENT
  MACHINE
  MACHINE_PART
  MEASUREMENT_POINT
  MEASUREMENT_DIRECTION
}

Enum device_type {
  MOBILE
  DATA_COLLECTOR
}

Enum speed_mode {
  FIXED
  VARIABLE
  UNKNOWN
}

Enum acquisition_mode {
  ROOT
  OFF_ROOT_FREE
}

Enum measurement_quality {
  PENDING
  VALID
  INCOMPLETE
  INVALID
}

Enum signal_quality {
  VALID
  MISSING
  CLIPPED
  CORRUPTED
  OUT_OF_RANGE
}

Enum temperature_status {
  VALID
  UNAVAILABLE
  INVALID
  OUT_OF_RANGE
}

Enum transfer_state {
  PENDING
  SELECTED
  UPLOADING
  FAILED
  COMMITTED
}

Enum processing_state {
  PENDING
  RUNNING
  READY
  FAILED
}

Enum screening_severity {
  NORMAL
  WARNING
  CRITICAL
  INSUFFICIENT_DATA
}

Enum threshold_kind {
  ISO
  USER_DEFINED
  TREND_DEVIATION
  HISTORICAL
  SIMILAR_MACHINE
}

Table factories {
  id uuid [pk]
  name varchar(200) [not null]
  code varchar(80) [not null, unique]
  authority_id uuid [not null, note: 'Current Desktop/Site authority']
  authority_epoch bigint [not null, default: 1]
  tree_revision bigint [not null, default: 0]
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
}

// One self-referencing table represents the entire fixed hierarchy:
// Line -> Department -> Machine -> Part -> Point -> Direction.
Table asset_nodes {
  id uuid [pk]
  factory_id uuid [not null, ref: > factories.id]
  parent_id uuid [ref: > asset_nodes.id]
  node_type asset_node_type [not null]
  name varchar(200) [not null]
  code varchar(80)
  sort_order int [not null, default: 0]
  is_active boolean [not null, default: true]
  revision bigint [not null, default: 1]
  deleted_at timestamptz
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (factory_id, parent_id, sort_order)
    (factory_id, node_type)
    (parent_id, node_type, name) [unique]
  }

  Note: 'The application/database trigger validates allowed parent-child types and prevents cycles. Direction nodes are stable Measurement Node IDs.'
}

// Searchable machine identity stays relational; optional technical fields are JSON.
Table machines {
  node_id uuid [pk, ref: - asset_nodes.id]
  factory_id uuid [not null, ref: > factories.id]
  tag varchar(120) [not null]
  priority smallint
  power_kw decimal(12,3)
  foundation_type varchar(100)
  machine_group varchar(120)
  speed_mode speed_mode [not null, default: 'UNKNOWN']
  nominal_rpm decimal(12,3)
  default_collection_duration_ms int
  image_uri text
  technical_data jsonb
  notes text

  indexes {
    (factory_id, tag) [unique]
    (machine_group, priority)
  }

  Note: 'node_id must point to a MACHINE node. points_count is derived and is not stored.'
}

// Used for Machine and Measurement Point technical history.
Table asset_spec_revisions {
  id uuid [pk]
  asset_node_id uuid [not null, ref: > asset_nodes.id]
  revision_no int [not null]
  spec_data jsonb [not null, note: 'Bearing, RPM, gear teeth, blade count and other type-specific fields']
  source varchar(40) [not null, default: 'MANUAL']
  effective_from timestamptz [not null]
  effective_to timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (asset_node_id, revision_no) [unique]
    (asset_node_id, effective_from)
  }
}

Table asset_templates {
  id uuid [pk]
  factory_id uuid [not null, ref: > factories.id]
  name varchar(200) [not null]
  template_type varchar(40) [not null, note: 'MACHINE, POINT or BRANCH']
  template_data jsonb [not null]
  version int [not null, default: 1]
  is_active boolean [not null, default: true]
  created_at timestamptz [not null, default: `now()`]

  indexes { (factory_id, name, version) [unique] }
}

Table devices {
  id uuid [pk]
  factory_id uuid [not null, ref: > factories.id]
  device_type device_type [not null]
  device_key varchar(160) [not null, note: 'Mobile installation ID or Data Collector serial number']
  display_name varchar(200)
  axis_count smallint
  app_or_firmware_version varchar(80)
  calibration_revision varchar(80)
  capabilities jsonb
  last_seen_at timestamptz
  revoked_at timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes { (factory_id, device_type, device_key) [unique] }
}

Table collection_assignments {
  id uuid [pk]
  factory_id uuid [not null, ref: > factories.id]
  mobile_device_id uuid [not null, ref: > devices.id]
  name varchar(200) [not null]
  tree_revision bigint [not null]
  status varchar(40) [not null, default: 'ACTIVE']
  created_at timestamptz [not null, default: `now()`]
  completed_at timestamptz

  indexes { (mobile_device_id, status) }
  Note: 'mobile_device_id must reference a MOBILE device.'
}

Table assignment_nodes {
  assignment_id uuid [not null, ref: > collection_assignments.id]
  measurement_node_id uuid [not null, ref: > asset_nodes.id]
  sort_order int [not null, default: 0]
  is_enabled boolean [not null, default: true]

  indexes {
    (assignment_id, measurement_node_id) [pk]
    (assignment_id, sort_order)
  }

  Note: 'measurement_node_id must reference a MEASUREMENT_DIRECTION node.'
}

// One durable Start -> Finish attempt. OFF_ROOT_LIVE is intentionally absent:
// its ring buffer is not persisted after the live view finishes.
Table measurements {
  id uuid [pk]
  factory_id uuid [not null, ref: > factories.id]
  assignment_id uuid [ref: > collection_assignments.id]
  measurement_point_id uuid [ref: > asset_nodes.id]
  mobile_device_id uuid [not null, ref: > devices.id]
  data_collector_id uuid [not null, ref: > devices.id]
  mode acquisition_mode [not null]
  measured_at timestamptz [not null]
  received_at timestamptz
  actual_rpm decimal(12,3)
  sample_rate_hz int [not null]
  expected_sample_count int [not null]
  duration_ms int [not null]
  acquisition_settings jsonb [not null]
  off_root_machine_tag varchar(120)
  spec_snapshot jsonb [not null, note: 'Immutable technical context at acquisition time']
  context_hash char(64) [not null]
  manifest_hash char(64) [not null]
  quality measurement_quality [not null, default: 'PENDING']
  quality_details jsonb
  transfer_status transfer_state [not null, default: 'PENDING']
  processing_status processing_state [not null, default: 'PENDING']
  temperature_value decimal(10,3)
  temperature_unit varchar(10) [not null, default: 'C']
  temperature_status temperature_status [not null, default: 'UNAVAILABLE']
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (factory_id, id) [unique]
    (measurement_point_id, measured_at)
    (mobile_device_id, transfer_status)
    (data_collector_id, measured_at)
  }

  Note: 'ROOT requires a Point and mapped Direction per signal. OFF_ROOT stays unassigned unless the user explicitly maps it.'
}

// One row per physical X/Y/Z channel. The raw bytes remain outside the DB.
Table measurement_signals {
  id uuid [pk]
  measurement_id uuid [not null, ref: > measurements.id]
  sensor_axis varchar(8) [not null]
  measurement_node_id uuid [ref: > asset_nodes.id, note: 'Mapping to a machine Direction in Root mode']
  storage_key text [not null, unique]
  sha256 char(64) [not null]
  byte_size bigint [not null]
  encoding varchar(60) [not null, note: 'Example: float32-le']
  compression varchar(40)
  sample_count int [not null]
  acceleration_unit varchar(20) [not null, default: 'm/s2']
  scale_factor decimal(20,10)
  quality signal_quality [not null]
  quality_details jsonb

  indexes {
    (measurement_id, sensor_axis) [unique]
    (measurement_node_id, measurement_id)
    sha256
  }
}

Table import_receipts {
  id uuid [pk]
  measurement_id uuid [not null, ref: - measurements.id]
  authority_id uuid [not null]
  authority_epoch bigint [not null]
  manifest_hash char(64) [not null]
  commit_sequence bigint [not null]
  committed_at timestamptz [not null]
  signature text

  indexes { (authority_id, authority_epoch, commit_sequence) [unique] }
  Note: 'Only this durable receipt permits user-confirmed cleanup on Mobile.'
}

Table derived_features {
  id uuid [pk]
  measurement_signal_id uuid [not null, ref: > measurement_signals.id]
  algorithm_name varchar(120) [not null]
  algorithm_version varchar(80) [not null]
  parameters_hash char(64) [not null]
  input_hash char(64) [not null]
  feature_values jsonb [not null, note: 'RMS, P2P, crest, kurtosis, overall and derived-data references']
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (measurement_signal_id, algorithm_name, algorithm_version, parameters_hash) [unique]
  }
}

Table threshold_rules {
  id uuid [pk]
  factory_id uuid [not null, ref: > factories.id]
  target_node_id uuid [ref: > asset_nodes.id]
  kind threshold_kind [not null]
  name varchar(200) [not null]
  ruleset_version varchar(80) [not null]
  rule_config jsonb [not null]
  is_active boolean [not null, default: true]
  effective_from timestamptz [not null]
  effective_to timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (factory_id, kind, is_active)
    (target_node_id, is_active)
  }
}

Table screening_results {
  id uuid [pk]
  measurement_id uuid [not null, ref: > measurements.id]
  measurement_node_id uuid [ref: > asset_nodes.id]
  ruleset_version varchar(80) [not null]
  context_hash char(64) [not null]
  severity screening_severity [not null]
  reasons jsonb [not null]
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (measurement_id, measurement_node_id, ruleset_version, context_hash) [unique]
    (measurement_node_id, created_at)
    (severity, created_at)
  }
}

Table audit_events {
  id uuid [pk]
  factory_id uuid [not null, ref: > factories.id]
  actor_id uuid
  entity_type varchar(80) [not null]
  entity_id uuid [not null]
  operation varchar(80) [not null]
  revision bigint
  change_data jsonb
  reason text
  occurred_at timestamptz [not null, default: `now()`]

  indexes { (factory_id, entity_type, entity_id, occurred_at) }
}

// PostgreSQL migration constraints not fully expressible in DBML:
// 1) validate the allowed node parent chain and prohibit cycles.
// 2) machines.node_id is a MACHINE; assignment/signal nodes are DIRECTIONS.
// 3) device roles must match their foreign-key use.
// 4) ROOT requires a Point and every mapped Direction belongs to that Point.
// 5) OFF_ROOT is not auto-attached by tag; OFF_ROOT_LIVE is never persisted.
// 6) axis_count is 1 or 3; rates/counts/durations/RPM/byte sizes are positive.
// 7) exactly one open-ended spec revision per asset node is allowed.
// 8) tree nodes with history are soft-deleted and never cascade-deleted.
