Enum signal_quality {
  VALID
  MISSING
  CLIPPED
  CORRUPTED
  OUT_OF_RANGE
}

Enum transfer_state {
  PENDING
  SELECTED
  UPLOADING
  FAILED
  COMMITTED
}

Table site {
  id uuid [pk]
  name varchar
  code varchar(80) [not null, unique]
  authority_id uuid [not null, note: 'Current Desktop/Site authority']
  authority_epoch bigint [not null, default: 1]
  tree_revision bigint [not null, default: 0]
  created_at timestamp
  updated_at timestamptz [not null, default: `now()`]
}

Table devices {
  id uuid [pk]
  site_id uuid [not null, ref: > site.id]
  device_type device_type [not null, note: 'MOBILE / DATA COLLECTOR']
  device_key varchar(160) [not null, note: 'Mobile installation ID or Data Collector serial number']
  display_name varchar(200)
  axis_count smallint
  app_or_firmware_version varchar(80)
  calibration_revision varchar(80)
  capabilities jsonb
  last_seen_at timestamptz
  revoked_at timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes { (site_id, device_type, device_key) [unique] }
} 

Table asset_nodes {
  id uuid [pk]
  site_id uuid [not null, ref: > site.id]
  parent_id uuid [ref: > asset_nodes.id]
  node_type asset_node_type [not null, note: 'Line - Department - Machine - Machine Part - Measurement Point - Mesurement Direction']
  name varchar(200) [not null]
  code varchar(80)
  sort_order int [not null, default: 0]
  is_active boolean [not null, default: true]
  revision bigint [not null, default: 1]
  deleted_at timestamptz
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (site_id, parent_id, sort_order)
    (site_id, node_type)
    (parent_id, node_type, name) [unique]
  }

  Note: 'The application/database trigger validates allowed parent-child types and prevents cycles. Direction nodes are stable Measurement Node IDs.'
}

Table asset_templates {
  id uuid [pk]
  site_id uuid [not null, ref: > site.id]
  name varchar(200) [not null]
  template_type varchar(40) [not null, note: 'MACHINE OR POINT']
  template_data jsonb [not null]
  version int [not null, default: 1]
  is_active boolean [not null, default: true]
  created_at timestamptz [not null, default: `now()`]

  indexes { (site_id, name, version) [unique] }
}

Table asset_spec_revisions {
  id uuid [pk]
  asset_node_id uuid [not null, ref: > asset_nodes.id]
  revision_no int [not null]
  spec_data jsonb [not null, note: 'Bearing, RPM, gear teeth, blade count and other type-specific fields']
  source varchar(40) [not null, default: 'MANUAL']
  effective_from timestamptz [not null]
  effective_to timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (asset_node_id, revision_no) [unique]
    (asset_node_id, effective_from)
  }
}

Table factory_lines {
  node_id uuid [pk, ref: - asset_nodes.id]
  site_id integer [not null]
  name varchar(200) [not null]
  created_at timestamp
  updated_at timestamptz [not null, default: `now()`]
}

Table departments {
  node_id uuid [pk, ref: - asset_nodes.id]
  site_id integer [not null]
  name varchar
}

Table machines {
  node_id uuid [pk, ref: - asset_nodes.id]
  site_id integer [not null]
  tag varchar(120) [not null]
  priority smallint
  power_kw decimal(12,3)
  foundation_type varchar(100)
  machine_group varchar(120)
  speed_mode speed_mode [not null, default: 'UNKNOWN']
  nominal_rpm decimal(12,3)
  default_collection_duration_ms int
  image_uri text
  technical_data jsonb
  notes text

  indexes {
    (site_id, tag) [unique]
    (machine_group, priority)
  }
}

Table machine_parts {
  node_id uuid [pk, ref: - asset_nodes.id]
  site_id integer [not null]
  name varchar
  critical bool
  machine_id integer [not null]
}

Table points {
  node_id uuid [pk, ref: - asset_nodes.id]
  site_id integer [not null]
  machine_type varchar
  RPM integer 
  bearing_type varchar 
}

Table directions {
  node_id uuid [pk, ref: - asset_nodes.id]
  site_id integer [not null]
  name varchar
}

Table collection_assignments {
  id uuid [pk]
  site_id uuid [not null, ref: > site.id]
  mobile_device_id uuid [not null, ref: > devices.id]
  name varchar(200) [not null]
  tree_revision bigint [not null]
  status varchar(40) [not null, note: 'ACTIVE, COMPLETED, CANCELLED, EXPIRED']
  created_at timestamptz [not null, default: `now()`]
  completed_at timestamptz

  indexes { (mobile_device_id, status) }
  Note: 'mobile_device_id must reference a MOBILE device.'
}

Table assignment_nodes {
  assignment_id uuid [not null, ref: > collection_assignments.id]
  measurement_node_id uuid [not null, ref: > asset_nodes.id, note: 'MEASUREMENT_DIRECTION node']
  sort_order int [not null, default: 0]
  is_enabled boolean [not null, default: true]

  indexes {
    (assignment_id, measurement_node_id) [pk]
    (assignment_id, sort_order)
  }
}

Table measurements {
  id uuid [pk]
  site_id uuid [not null, ref: > site.id]
  assignment_id uuid [ref: > collection_assignments.id, note: 'IF ROUTE']
  measurement_point_id uuid [ref: > asset_nodes.id]
  mobile_device_id uuid [not null, ref: > devices.id, note: 'MOBILE FROM COLLECTOR']
  data_collector_id uuid [not null, ref: > devices.id, note: 'IMPLANTED COLLECTOR']
  mode acquisition_mode [not null, note: 'ROUTE / OFF ROUTE']
  measured_at timestamptz [not null]
  received_at timestamptz
  actual_rpm decimal(12,3)
  sample_rate_hz int [not null]
  duration_ms int [not null]
  spec_snapshot jsonb [not null, note: 'Immutable technical context at acquisition time']
  context_hash char(64) [not null]
  manifest_hash char(64) [not null]
  transfer_status transfer_state [not null, default: 'PENDING']
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (site_id, id) [unique]
    (measurement_point_id, measured_at)
    (mobile_device_id, transfer_status)
    (data_collector_id, measured_at)
  }

  Note: 'ROOT requires a Point and mapped Direction per signal. OFF_ROOT stays unassigned unless the user explicitly maps it.'
}

Table measurement_signals {
  id uuid [pk]
  measurement_id uuid [not null, ref: > measurements.id]
  sensor_axis varchar(8) [not null, note: 'OFF-ROUTE']
  measurement_node_id uuid [ref: > asset_nodes.id, note: 'Mapping to a machine Direction in Root mode']
  storage_key text [not null, unique, note: 'local file address']
  sha256 char(64) [not null]
  byte_size bigint [not null]
  encoding varchar(60) [not null, note: 'Example: float32-le']
  compression varchar(40)
  sample_count int [not null]
  acceleration_unit varchar(20) [not null, default: 'm/s2']
  quality signal_quality [not null]

  indexes {
    (measurement_id, sensor_axis) [unique]
    (measurement_node_id, measurement_id)
    sha256
  }
}
